function(extract_production start_text end_text output)
  string(FIND "${production}" "${start_text}" first)
  string(FIND "${production}" "${end_text}" last)
  if(first LESS 0 OR last LESS first)
    message(FATAL_ERROR "Production callback boundaries changed: ${start_text}")
  endif()
  math(EXPR length "${last} - ${first}")
  string(SUBSTRING "${production}" ${first} ${length} body)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/${output}" "${body}")
endfunction()
